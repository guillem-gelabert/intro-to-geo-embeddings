```python
RELEASE_BASE = "https://github.com/{org}/{repo}/releases/download/{tag}"

def load(area: str, year: int) -> gpd.GeoDataFrame:
    local = Path(f"data/{area}/{area}_aef_{year}_v1.1.parquet")
    if local.exists():
        return gpd.read_parquet(local)
    manifest = _load_manifest()
    entry = manifest[local.name]
    url = f"{RELEASE_BASE...}/{local.name}"
    resp = requests.get(url, timeout=60)
    resp.raise_for_status()
    sha = hashlib.sha256(resp.content).hexdigest()
    assert sha == entry["sha256"], f"checksum mismatch for {local.name}"
    local.parent.mkdir(parents=True, exist_ok=True)
    local.write_bytes(resp.content)
    return gpd.read_parquet(local)
```
`manifest.json` (produced by `pipeline/write.py` in Plan 02-09) has shape `{"<filename>": {"sha256": "...", "rows": N, "built_at": "...", "builder_git_sha": "...", "sources": {...}}}`.
Provenance is written into Parquet key-value metadata under `b"course_provenance"` (Plan 02-09, ARCHITECTURE.md Pattern 5) — `load()` must read it back via `pyarrow.parquet.read_table(path).schema.metadata[b"course_provenance"]`, `json.loads(...)`, and assign to `gdf.attrs["course_provenance"]` after `gpd.read_parquet` (geopandas' own read does not surface arbitrary custom metadata into `.attrs` automatically — read it explicitly with pyarrow).
